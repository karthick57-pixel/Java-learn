Even or Odd:

import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter a Number: ");
        int number = sc.nextInt();
        if(number % 2 == 0){
               System.out.println("Even");
        }
        else{
        System.out.println("Odd");
    }
}
}
